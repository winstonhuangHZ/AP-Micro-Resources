\sessionfooter{H01}{15–22 min · Core model}
\begin{frame}[t]{Scarcity turns resources into trade-offs}
  \tagline{H01 · SCARCITY, TRADE-OFF, OPPORTUNITY COST}
  \vspace{0.10cm}
  \begin{columns}[T,totalwidth=\textwidth]
    \column{0.47\textwidth}
      \callout{PaleBlue}{%
        {\color{APBlue}\fontsize{20}{24}\selectfont\bfseries Scarcity}\par\vspace{0.06cm}
        Available resources are limited relative to desired uses.
      }
      \vspace{0.10cm}
      {\color{Muted}\fontsize{15}{18}\selectfont
      The resource may be land, labor, physical capital, time, or productive capacity.}
    \column{0.47\textwidth}
      \callout{PaleOrange}{%
        {\color{APOrange}\fontsize{20}{24}\selectfont\bfseries Trade-off}\par\vspace{0.06cm}
        Choosing one use means giving up another use.
      }
      \vspace{0.10cm}
      {\color{Ink}\fontsize{16}{20}\selectfont
      \highlight{Opportunity cost} = the value of the best alternative given up.}
  \end{columns}
  \vspace{0.10cm}
  {\color{Muted}\centering\fontsize{13}{16}\selectfont\sffamily
    scarce resource $\longrightarrow$ choice $\longrightarrow$ foregone alternative\par}
\end{frame}
